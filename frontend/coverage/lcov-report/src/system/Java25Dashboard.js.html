import React, { useState, useEffect } from 'react';
import { apiClient, extractErrorMessage } from '../api';
import { useToast } from '../ToastContext';
 
export default function Java25Dashboard() {
  const { addToast } = useToast();
 
  // Diagnostics state
  const [nativeInfo, setNativeInfo] = useState(null);
  const [payloadText, setPayloadText] = useState('Modern Java 25 Native FFM API Payload');
  const [loadingNative, setLoadingNative] = useState(false);
 
  // Virtual threads state
  const [vtResults, setVtResults] = useState([]);
  const [runningVt, setRunningVt] = useState(false);
  const [customTasks, setCustomTasks] = useState('MatriculaBatch, CalculoPromedios, SincronizacionCloud, NotificacionesPush');
 
  // Audit log state
  const [auditLog, setAuditLog] = useState([]);
  const [loadingAudit, setLoadingAudit] = useState(false);
 
  useEffect(() => {
    fetchNativeInfo();
    fetchAuditLog();
  }, []);
 
  async function fetchNativeInfo() {
    setLoadingNative(true);
    try {
      const res = await apiClient.get(`/system/native-info?payload=${encodeURIComponent(payloadText)}`);
      setNativeInfo(res.data);
    } catch (err) {
      addToast(extractErrorMessage(err), 'error');
    } finally {
      setLoadingNative(false);
    }
  }
 
  async function runVirtualThreads() {
    setRunningVt(true);
    try {
      const taskList = customTasks
        .split(',')
        .map(t => t.trim())
        .filter(Boolean);
 
      const res = await apiClient.post('/system/virtual-threads/run', taskList);
      setVtResults(res.data || []);
      addToast(`Ejecutadas ${res.data?.length || 0} tareas en Hilos Virtuales`, 'success');
      // Refresh audit log as tasks may emit events
      fetchAuditLog();
    } catch (err) {
      addToast(extractErrorMessage(err), 'error');
    } finally {
      setRunningVt(false);
    }
  }
 
  async function fetchAuditLog() {
    setLoadingAudit(true);
    try {
      const res = await apiClient.get('/courses/audit-log');
      setAuditLog(res.data || []);
    } catch (err) {
      // Endpoint may be empty if no events yet
      console.warn('Audit log fetch:', err);
    } finally {
      setLoadingAudit(false);
    }
  }
 
  return (
    <div>
      {/* Top Banner */}
      <div className="glass-card mb-4" style={{ background: 'linear-gradient(135deg, rgba(99, 102, 241, 0.15), rgba(6, 182, 212, 0.1))', border: '1px solid rgba(99, 102, 241, 0.3)' }}>
        <div className="d-flex justify-content-between align-items-center flex-wrap gap-3">
          <div>
            <div className="d-flex align-items-center gap-2 mb-1">
              <span className="brand-pill">Java 25 Innovation Lab</span>
              <span className="status-indicator online">
                <span className="status-dot"></span>
                <span>OpenJDK 25.0.1 Active</span>
              </span>
            </div>
            <h3 style={{ margin: 0, fontWeight: 700, color: '#f8fafc' }}>
              Diagnóstico de Capacidades Modernas del Backend
            </h3>
            <p className="text-muted small mb-0 mt-1">
              Inspección en tiempo real de Virtual Threads (Loom), Foreign Function & Memory (FFM) y Pattern Matching con Records.
            </p>
          </div>
 
          <div className="d-flex gap-2">
            <button className="btn-modern-secondary" onClick={() => { fetchNativeInfo(); fetchAuditLog(); }}>
              Refrescar Todo
            </button>
          </div>
        </div>
      </div>
 
      {/* Row 1: FFM API / Native Bridge */}
      <div className="glass-card mb-4">
        <div className="glass-card-header">
          <h4 className="glass-card-title">
            <span>Foreign Function & Memory (FFM) API</span>
            <span className="brand-pill">Java 25 Off-Heap</span>
          </h4>
          <button className="btn-modern-cyan" onClick={fetchNativeInfo} disabled={loadingNative}>
            {loadingNative ? 'Consultando...' : 'Re-calcular en Memoria Nativa'}
          </button>
        </div>
 
        <div className="mb-3">
          <label className="form-label text-muted small">Payload para Digest Off-Heap (Arena Allocator):</label>
          <div className="input-group">
            <input
              className="form-control modern-input"
              value={payloadText}
              onChange={e => setPayloadText(e.target.value)}
              placeholder="Escribe un texto para procesar fuera de la heap de la JVM"
            />
          </div>
        </div>
 
        {nativeInfo ? (
          <div className="row g-3">
            <div className="col-md-3 col-sm-6">
              <div className="metric-card">
                <div className="metric-label">Dirección Nativa Off-Heap</div>
                <div className="metric-value" style={{ fontSize: '1.1rem', color: '#67e8f9' }}>
                  {nativeInfo.nativeMemoryAddress}
                </div>
                <div className="metric-sub">{nativeInfo.allocatedBytes} bytes asignados</div>
              </div>
            </div>
 
            <div className="col-md-3 col-sm-6">
              <div className="metric-card">
                <div className="metric-label">Proceso & Arquitectura</div>
                <div className="metric-value" style={{ fontSize: '1.2rem', color: '#a5b4fc' }}>
                  PID {nativeInfo.processId}
                </div>
                <div className="metric-sub">{nativeInfo.operatingSystem} ({nativeInfo.osArch})</div>
              </div>
            </div>
 
            <div className="col-md-6 col-sm-12">
              <div className="metric-card">
                <div className="metric-label">Digest Criptográfico Off-Heap (SHA-256)</div>
                <div className="font-monospace small text-truncate" style={{ color: '#10b981', background: 'rgba(0,0,0,0.3)', padding: '6px 10px', borderRadius: 6 }}>
                  {nativeInfo.sha256OffHeapDigest}
                </div>
                <div className="metric-sub">Calculado utilizando MemorySegment y SegmentAllocator nativo</div>
              </div>
            </div>
          </div>
        ) : (
          <div className="text-center py-3 text-muted">Cargando diagnóstico nativo...</div>
        )}
      </div>
 
      {/* Row 2: Virtual Threads Benchmark */}
      <div className="glass-card mb-4">
        <div className="glass-card-header">
          <h4 className="glass-card-title">
            <span>Project Loom - Hilos Virtuales</span>
            <span className="brand-pill">Concurrencia Ligera</span>
          </h4>
          <button className="btn-modern-primary" onClick={runVirtualThreads} disabled={runningVt}>
            {runningVt ? 'Ejecutando en Loom...' : 'Disparar Tareas Concurrentes'}
          </button>
        </div>
 
        <div className="mb-3">
          <label className="form-label text-muted small">Lista de Tareas Concurrentes (separadas por coma):</label>
          <input
            className="form-control modern-input"
            value={customTasks}
            onChange={e => setCustomTasks(e.target.value)}
          />
        </div>
 
        {vtResults.length > 0 && (
          <div className="table-responsive">
            <table className="table table-dark table-hover mb-0" style={{ background: 'transparent' }}>
              <thead>
                <tr className="text-muted small" style={{ borderBottom: '1px solid var(--border-color)' }}>
                  <th>Tarea</th>
                  <th>Hilo JVM (VirtualThread)</th>
                  <th>Duración</th>
                  <th>Estado</th>
                </tr>
              </thead>
              <tbody>
                {vtResults.map((r, i) => (
                  <tr key={i} style={{ borderBottom: '1px solid rgba(255, 255, 255, 0.05)' }}>
                    <td style={{ fontWeight: 600, color: '#f8fafc' }}>{r.taskName}</td>
                    <td>
                      <span className="badge-glow-indigo font-monospace small">
                        {r.threadName}
                      </span>
                    </td>
                    <td className="font-monospace" style={{ color: '#67e8f9' }}>{r.durationMs} ms</td>
                    <td>
                      <span className="badge-role student">COMPLETADO</span>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}
      </div>
 
      {/* Row 3: Pattern Matching & Audit Log */}
      <div className="glass-card">
        <div className="glass-card-header">
          <h4 className="glass-card-title">
            <span>Auditoría Académica (Record Patterns)</span>
            <span className="brand-pill">Pattern Matching Switch</span>
          </h4>
          <button className="btn-modern-secondary" onClick={fetchAuditLog} disabled={loadingAudit}>
            {loadingAudit ? 'Cargando...' : 'Actualizar Log'}
          </button>
        </div>
 
        {auditLog.length === 0 ? (
          <div className="p-3 text-center text-muted">
            No hay eventos registrados en el búfer de auditoría en memoria aún. Interactúa matriculando alumnos o creando cursos para generar eventos.
          </div>
        ) : (
          <div className="code-console">
            {auditLog.map((log, index) => (
              <div key={index} style={{ marginBottom: 6, color: '#cbd5e1' }}>
                <span style={{ color: '#06b6d4', marginRight: 8 }}>[{index + 1}]</span>
                <span>{log}</span>
              </div>
            ))}
          </div>
        )}
      </div>
    </div>
  );
}
